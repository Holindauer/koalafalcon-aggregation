\section{Preliminaries}

\subsection{Notation}

Let $n$ be a power-of-two, $q$ is prime, $R = \mathbb{Z}[X]/(X^n + 1)$, and $R_q = R/qR$. 
In the context of Falcon, we will refer to $\alpha \in \mathbb{R}^{>1}$ as the trapdoor 
basis quality, $\beta \in \mathbb{R}^{>0}$ as the signature norm bound, $k \in \mathbb{N}$
as the salt bit length,  $s \in \mathbb{R}^{>0}$ as the signature Gaussian standard deviation, 
tailcut rate $\tau > 1$, $\epsilon > 0$, security parameter $\lambda$, max number of signing 
queries $Q_s \in \mathbb{N}$, max number of random oracle queries $Q_H \in \mathbb{N}$,
max number of signing attempts $C_s \in \mathbb{N}$, and R\'enyi orders $a_u$ and $a_p$.
For a lattice basis $B$, write $\|B\|_{\mathrm{GS}}$ for its Gram--Schmidt norm.
For $x \in \mathbb{R}^m$, write $\|x\|_2$ for its $\ell_2$ norm. Write $U(R_q)$ for the uniform 
distribution on $R_q$. Write $D_{\Lambda,s}$ for the discrete Gaussian on a lattice $\Lambda$ 
with width $s$. Write $\eta_\epsilon(\Lambda)$ for the smoothing parameter of $\Lambda$ at 
slack $\epsilon$. Write $R_a(P \,\|\, Q)$ for R\'enyi divergence of order $a$ between 
distributions $P$ and $Q$. We will sometimes refer to elements of $R, R_q$ as matrices in 
reference to their $(-1)$-skew-circulant representation, which preserves the commutativity 
of cyclotomic ring multiplication.

\subsection{Definitions}

\begin{definition}[Theorem~2.9~\cite{Menezes2026GentleIntro}]
Let $\Lambda = \Lambda(B_1)$ be a lattice with basis $B_1$.
A matrix $B_2$ is also a basis of $\Lambda$ if and only if there exists a unimodular matrix $U$ with $\det U = \pm 1$ such that $UB_2 = B_1$.
\end{definition}

\begin{definition}[Regev, \S3~\cite{Regev2009LatticeSurvey}]
Let $q$ be prime, $A \in \mathbb{Z}_q^{n \times m}$ have linearly independent rows, and $m \ge n$.
The $m$-dimensional $q$-ary orthogonal lattice associated with $A$ is
\[
\Lambda_q^\perp(A) = \{y \in \mathbb{Z}^m : Ay = 0 \pmod q\}.
\]
Its determinant is $\det(\Lambda_q^\perp(A)) = q^n$.
\end{definition}

\begin{definition}[Definition~2.11~\cite{Menezes2026GentleIntro}]
The volume of a lattice is its determinant and measures the volume of its fundamental parallelepiped.
\end{definition}

\begin{definition}[Definition~2.12~\cite{Menezes2026GentleIntro}]
The $i$-th successive minimum of an $m$ dimensional lattice $\lambda_i \in \{\lambda_1, ..., \lambda_m\}$ is the smallest real number $r$ such that 
the lattice contains $i$ linearly independent vectors at most length $r$. Intuitively, each $\lambda_i$ is the smallest radius of an $m$ dimensional 
ball centered at the origin such that it contains each of these $i$ linearly independent vectors of the lattice.
\end{definition}

\begin{definition}[Regev, \S3~\cite{Regev2009LatticeSurvey}; \cite{Chen2016GaussianHeuristic}]
For $\Lambda_q^\perp(A)$,
\[
\lambda_1(\Lambda_q^\perp(A)) \approx q^{n/m}\sqrt{\frac{m}{2\pi e}}.
\]
\end{definition}

\begin{definition}[Definition~10~\cite{Fouque2024CoreFalcon}; Regev, \S3~\cite{Regev2009LatticeSurvey}]
For $h \in R_q \setminus \{0\}$ and $c \in R_q$, define
\[
\Lambda_h = \{(u,v) \in R^2 : u + vh = 0 \pmod q\},
\qquad
\Lambda_{h,c} = \{(u,v) \in R^2 : u + vh = c \pmod q\}.
\]
Then $\Lambda_{h,c} = \Lambda_h + t$ for some shift $t$ is called the shifted lattice (or lattice coset) of $\Lambda_h$, and corresponds to a translation by $t$.
With $m = 2n$,

\[
\lambda_1(\Lambda_h) \approx q^{n/m}\sqrt{\frac{m}{2\pi e}}
= \sqrt{\frac{qn}{\pi e}}.
\]
\end{definition}

\begin{definition}[Definition~10~\cite{Fouque2024CoreFalcon}]
In the $(Q_H+1)$-R-ISIS game, the adversary receives $(h, c_1, \ldots, c_t)$ with $h \leftarrow \mathrm{TpdGen}(\alpha,q)$, each $c_i \leftarrow U(R_q)$, and $t = Q_H + 1$.
It must output a short pair $(u,v) \in R^2$ and an index $j$ such that $c_j = u + vh \pmod q$ and $\|(u,v)\|_2 \le B$.
\end{definition}

\begin{definition}[Lemma~3.1~\cite{GPV2008}]
For any $n$-dimensional lattice $\Lambda$ and real $\epsilon > 0$
$$
\eta_\epsilon(\Lambda_{h,q}) \le \eta_\epsilon(\mathbb{Z}^{2n}) \cdot ||B||_{GS, min}.
$$

Where $||B||_{GS, min}$ is the Gram-Schmidt minimum across all bases $B$ of $\Lambda$.

\end{definition}

\begin{definition}[Theorem~2.9~\cite{Menezes2026GentleIntro}; \cite{Fouque2024CoreFalcon}]
Let $q$ be prime, $n = 2^\kappa$, $\phi = X^n + 1$ split over $\mathbb{Z}_q$, and $f,g \in R_q^\times$ with
\[
h \equiv gf^{-1} \pmod{\phi,q},
\qquad
fG - gF \equiv q \pmod{\phi}.
\]
The private and public bases of the NTRU lattice are
\[
B_1 =
\begin{bmatrix}
g & -f \\
G & -F
\end{bmatrix},
\qquad
B_2 =
\begin{bmatrix}
-h & I_n \\
qI_n & O_n
\end{bmatrix},
\]
with ring elements represented as $(-1)$-skew-circulant matrices.
\end{definition}

\begin{definition}[Lemma~4~\cite{Fouque2024CoreFalcon}; Lemma~3.1~\cite{GPV2008}]
CoreFalcon+ sets
\[
\eta_\epsilon^{\mathrm{upper}}(\mathbb{Z}^{2n})
= \frac{1}{\pi}\sqrt{\frac{\ln\!\left(4n\left(1+\frac{1}{\epsilon}\right)\right)}{2}},
\qquad
s = \eta_\epsilon^{\mathrm{upper}}(\mathbb{Z}^{2n}) \cdot \alpha \sqrt{q},
\qquad
\beta = \tau s \sqrt{2n}.
\]
Trapdoor generation satisfies $\|B\|_{\mathrm{GS}} \le \alpha\sqrt{q}$.
The smoothing condition used by the reduction is $s \ge \eta_\epsilon(\Lambda_{h,q})$; GPV, Lem.~3.1~\cite{GPV2008} gives $s \ge \eta_\epsilon(\mathbb{Z}^{2n}) \cdot \|B_{f,g}\|_{\mathrm{GS}}$.
\end{definition}

\begin{corollary}[Corollary~2~\cite{Fouque2024CoreFalcon}]\label{cor:ru-renyi}
Let \(q\) be prime, \(h \in R_q \setminus \{0\}\), \(a \in (1,\infty)\),
\(\epsilon \in (0, \frac{1}{2})\),
\(s \ge \eta_{\epsilon}(\Lambda_{h,q})\), \(P = U(R_q)\), and \(Q\) the
distribution of \(u + v \cdot h \mod q\) with \(u, v \sim D_{R,s}\).
Then
\[
R_a(P \,\|\, Q)
\lesssim
1 +
\frac{2a\epsilon^2}{(1-\epsilon)^2}.
\]
\end{corollary}

\begin{corollary}[Corollary~1~\cite{Fouque2024CoreFalcon}]\label{cor:rp-renyi}
Let \(n\) be a positive integer, \(a \in (1 , \infty)\),
\(\epsilon \in (0, 1/4)\). Then for \(PreSmp := \text{FFO Sampler}\),
and lattice \(\Lambda = \Lambda(B)_{(c, 0)}\) for any basis
\(B \in \mathbb{Z}^{2n \times 2n}\),
\(s \ge \eta_\epsilon(\mathbb{Z}^{2n}) \cdot ||B||_{GS}\), and arbitrary
syndrome \(c \in R_q\), the Rényi divergence is bounded by
\[
R_a\!\left(
\mathrm{PreSmp}(B,s,(c,0)) \,\middle\|\, D_{\Lambda,s}
\right)
\lesssim 1 + 2a\epsilon^2.
\]
\end{corollary}

\begin{lemma}[Lemma~7~\cite{Fouque2024CoreFalcon}]\label{lem:ffo-relerror}
Let \(n \in \mathbb{Z}^+\),
\(\epsilon \in (0, 1/4)\). Then the relative error of
\(PreSmp := \text{FFO Sampler}\) and lattice \(\Lambda = \Lambda(B)\)
for any basis \(B \in \mathbb{Z}^{2n \times 2n}\),
\(s \ge \eta_\epsilon(\mathbb{Z}^{2n \times 2n}) \cdot ||B||_{GS}\), and
arbitrary syndrome \(c \in R_q\) is bounded by
\[
\delta_{\mathrm{RE}}\!\left(
    \mathrm{PreSmp}(B,s,(c,0)),\, D_{\Lambda,s}
\right)
\le
\left(
    \frac{1+\epsilon/(2n)}{1-\epsilon/(2n)}
\right)^{2n}
- 1
\approx 2\epsilon 
\]
\end{lemma}

\begin{lemma}[Lemma~10~\cite{Fouque2024CoreFalcon}]\label{lem:relerror-tail}
Let \(P\), \(Q\) be two distributions
sharing the same space of outputs with non-zero probability
\(\sup(Q) = \sup(P) = \mathbb{Z}^{n}\) and
\(\delta_{RE}(P, Q) = \delta\). Then for any \(\beta \ge 0\)
\[
\Pr_{x \leftarrow P}\!\left[\|x\|_2 > \beta\right]
\le
\Pr_{x \leftarrow Q}\!\left[\|x\|_2 > \beta\right] \cdot (1+\delta)
\]
\end{lemma}

\begin{lemma}[Lemma~2~\cite{Fouque2024CoreFalcon}]\label{lem:gauss-tail}
For any lattice
\(\Lambda \subseteq \mathbb{R}^n\), width \(s > 0\), shift
\(t \in \mathbb{R}^n\), and tailcut rate \(\tau > 1\)
\[
\Pr_{z \leftarrow D_{\Lambda + t,\, s}}
\!\left[
\|z\|_2 > \tau s \sqrt{n}
\right]
\le
\frac{\rho_s(\Lambda)}{\rho_s(\Lambda + t)}
\left(
\sqrt{e^{1 - \tau^2}\tau^2}
\right)^n 
\]
\end{lemma}

\begin{lemma}[Lemma~5~\cite{Fouque2024CoreFalcon}]\label{lem:gauss-mass}
For any \(n\)-dimensional lattice \(\Lambda\), center
\(c \in \mathbb{R}^n\), and reals \(0 < \epsilon < 1\), and
\(s \ge \eta_\epsilon(\Lambda)\), we have
\[
\rho_{s,c}(\Lambda)
\in
[\,1-\epsilon,\;1+\epsilon\,]
\cdot
\frac{(\sqrt{2\pi} \cdot s)^n}{\det(\Lambda)}
\]
\end{lemma}

\begin{definition}[Theorem~1~\cite{Fouque2024CoreFalcon}]

  $$
  \begin{aligned}
  \operatorname{Adv}^{Q_s\text{-}\mathrm{UF}\text{-}\mathrm{CMA}}_{\mathrm{CoreFalcon}^{+}, A}
  &\le
  \left(
  r_u^{C_s}
  \cdot
  \left(
  r_p^{C_s}
  \cdot
  \operatorname{Adv}^{(Q_H+1)\text{-}\mathrm{R}\text{-}\mathrm{ISIS}}_{q,\alpha,\beta,B}
  \right)^{\frac{a_p-1}{a_p}}
  \right)^{\frac{a_u-1}{a_u}}
  \\
  &\quad+
  \sum_{i=0}^{Q_s}
  \binom{C_s}{i}
  (1-p_{\mathrm{PreSmp},\beta})^{C_s-i}
  (p_{\mathrm{PreSmp},\beta})^i
  \quad+
  \frac{C_s(Q_H+C_s)}{2^k}
  \end{aligned}
  $$
  
  $$
  p_{\mathrm{PreSmp},\beta}
  :=
  \min_{(B,\cdot)\in \mathrm{TpdGen},\, c\in R_q}
  \Pr_{(s_1,s_2)\gets_{\$}\, \mathrm{PreSmp}(B,s,(c,0))}
  \left[
  \left\|(s_1,s_2)\right\|_2 \le \beta
  \right],
  $$
  
  $$
  r_u = \max_{(\cdot,h)\in \mathrm{TpdGen}} R_{a_u}(P \,\|\, Q_h),
  $$
  
  $$
  r_p = \max_{(B,\cdot)\in \mathrm{TpdGen},\, c\in R_q} R_{a_p} \left( \mathrm{PreSmp}(B,s,(c,0)) \,\middle\|\, D_{\Lambda,s} \right),
  $$

  Where $\operatorname{Adv}^{(Q_H+1)\text{-}\mathrm{R\text{-}ISIS}}_{q,\alpha,\beta,B}$ is the advantage adversary $A$ has in determining a 
  pre-image $(u, v) \in R^2$ satisfying $c_j = u + vh \mod q \land ||(u, v)||_2 \le B$ where $B > 0$ is the R-ISIS solution norm bound, 
  $c_j \in \{c_1, ..., c_t\}$ are uniformly sampled. $p_{\mathrm{PreSmp},\beta}$ is the worst-case probability of the FFO Sampler algorithm 
  correctly sampling an $(s_1, s_2) \in R^2$ such that $||(s_1,s_2)||_2 \le \beta$. $r_u$ is the multiplicative Rényi divergence 
  $R_{a_u}(P \,\|\, Q_h) \ge 1$ (order $a_u$) between $P = U(R_q)$ and $Q_h$, the distribution of $s_1 + s_2\cdot h \mod q$ for $s_1, s_2 \sim D_{R, s}$. 
  Values $r_u \approx 1$ indicate negligible loss, consistent with the factor $r_u^{C_s}$ in the bound. $r_p$ is the multiplicative Rényi divergence 
  $R_{a_p}(\mathrm{PreSmp}(B,s,(c,0)) \,\|\, D_{\Lambda,s}) \ge 1$ (order $a_p$) between FFO sampler output and the discrete Gaussian $D_{\Lambda,s}$, 
  where $\Lambda=\Lambda(B)_{(c,0)}$ is the shifted preimage coset for syndrome $(c,0)$; $r_p \approx 1$ is the good case, consistent with $r_p^{C_s}$. 
  $C_s$ is the max number of attempts to sample $(s_1, s_2)$ a valid signature satisfying $||(s_1,s_2)||_2 \le \beta$.
\end{definition}

\begin{definition}[Regev, \S3~\cite{Regev2009LatticeSurvey}; \cite{GamaNguyen2008}]
For a random $m$-dimensional $q$-ary lattice $\Lambda_q^\perp(A)$, lattice reduction with root-Hermite factor $\delta$ finds vectors of length approximately
\[
\min\{q,\, q^{n/m}\delta^m\}.
\]
Typical values are $\delta \approx 1.013$ for fast BKZ and $\delta \approx 1.011$ for slower, more precise variants.
\end{definition}
